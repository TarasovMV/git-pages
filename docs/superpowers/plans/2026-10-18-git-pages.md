"""Ключи страниц: валидация и разрешение ключа в файл."""
import re
from pathlib import Path

SEGMENT_RE = re.compile(r"^[a-z0-9][a-z0-9-]*$")
MAX_DEPTH = 4


def validate_key(key: str) -> list[str]:
    """Список ошибок ключа; пустой список = ключ валиден."""
    errors: list[str] = []
    if not key:
        return ["ключ пуст"]
    segments = key.split("/")
    if len(segments) > MAX_DEPTH:
        errors.append(f"больше {MAX_DEPTH} сегментов")
    for seg in segments:
        if not SEGMENT_RE.match(seg):
            errors.append(f"невалидный сегмент {seg!r}: нужно [a-z0-9][a-z0-9-]*")
    return errors


def page_file_candidates(repo_root: Path, key: str) -> list[Path]:
    """Кандидаты файла страницы в порядке приоритета: папка, затем файл."""
    return [
        repo_root / "pages" / Path(*key.split("/")) / "index.html",
        repo_root / "pages" / (key + ".html"),
    ]


def resolve_page_file(repo_root: Path, key: str) -> Path | None:
    """Существующий файл страницы по ключу или None."""
    for candidate in page_file_candidates(repo_root, key):
        if candidate.is_file():
            return candidate
    return None
```

"""Парсер блока page-meta — HTML-комментария с полями title/description."""
import re
from dataclasses import dataclass

META_RE = re.compile(r"<!--\s*page-meta\s*\n(.*?)-->", re.DOTALL)
FIELD_RE = re.compile(r"^([a-z-]+):[ \t]*(.*)$", re.MULTILINE)


@dataclass
class PageMeta:
    title: str = ""
    description: str = ""


def parse_page_meta(html_text: str) -> PageMeta:
    m = META_RE.search(html_text)
    if not m:
        return PageMeta()
    fields = dict(FIELD_RE.findall(m.group(1)))
    return PageMeta(
        title=fields.get("title", "").strip(),
        description=fields.get("description", "").strip(),
    )
```

`tools/pageslib/scanner.py`:

```python
"""Сканер pages/: находит страницы, ключи, метаданные и даты из git."""
import subprocess
from dataclasses import dataclass
from datetime import date
from pathlib import Path

from .meta import parse_page_meta

TODAY = date.today().isoformat()


@dataclass
class PageEntry:
    key: str            # например "shagal-electricity/panel"
    path: Path          # абсолютный путь к файлу страницы
    kind: str           # "folder" | "file"
    title: str
    description: str
    updated: str        # YYYY-MM-DD из git log
    created: str        # YYYY-MM-DD из git log

    def url_path(self) -> str:
        suffix = "/" if self.kind == "folder" else ""
        return f"/pages/{self.key}{suffix}"


def _key_of(pages_dir: Path, html_file: Path) -> tuple[str | None, str]:
    posix = html_file.relative_to(pages_dir).as_posix()
    if posix == "index.html":
        return None, ""  # pages/index.html сам по себе страницей не является
    if posix.endswith("/index.html"):
        return posix[: -len("/index.html")], "folder"
    return posix[: -len(".html")], "file"


def _git_dates(repo_root: Path, rel_path: str) -> tuple[str, str]:
    """(updated, created) из git log; без коммитов — TODAY."""
    base = ["git", "-C", str(repo_root), "log", "--format=%cs", "--", rel_path]
    updated = subprocess.run(base + ["-1"], capture_output=True, text=True, encoding="utf-8")
    created = subprocess.run(base + ["--diff-filter=A"], capture_output=True, text=True, encoding="utf-8")
    up = updated.stdout.strip() or TODAY
    added = [line for line in created.stdout.strip().splitlines() if line]
    return up, (added[-1] if added else TODAY)


def scan_pages(repo_root: Path) -> list[PageEntry]:
    pages_dir = repo_root / "pages"
    if not pages_dir.is_dir():
        raise FileNotFoundError(f"нет папки pages/ в {repo_root}")
    by_key: dict[str, PageEntry] = {}
    for html_file in sorted(pages_dir.rglob("*.html")):
        key, kind = _key_of(pages_dir, html_file)
        if key is None:
            continue
        meta = parse_page_meta(html_file.read_text(encoding="utf-8"))
        rel = html_file.relative_to(repo_root).as_posix()
        updated, created = _git_dates(repo_root, rel)
        entry = PageEntry(
            key=key, path=html_file, kind=kind,
            title=meta.title, description=meta.description,
            updated=updated, created=created,
        )
        existing = by_key.get(key)
        if existing is None or (existing.kind == "file" and kind == "folder"):
            by_key[key] = entry
    return list(by_key.values())
```

import unittest
from pathlib import Path

from pageslib.gallery import page_href, render_index
from pageslib.scanner import PageEntry


def entry(key: str, kind: str = "folder", title: str = "", description: str = "",
          updated: str = "2026-01-01", created: str = "2026-01-01") -> PageEntry:
    return PageEntry(key=key, path=Path("pages") / key / "index.html", kind=kind,
                     title=title, description=description, updated=updated, created=created)


class RenderIndexTests(unittest.TestCase):
    def test_links_and_titles(self):
        html = render_index([
            entry("a", title="Альфа", description="Первая"),
            entry("shagal/panel", kind="file", title="Панель"),
        ])
        self.assertIn('href="pages/a/"', html)
        self.assertIn('href="pages/shagal/panel"', html)
        self.assertIn("Альфа", html)
        self.assertIn("Первая", html)

    def test_title_falls_back_to_key(self):
        self.assertIn(">b</a>", render_index([entry("b")]))

    def test_escapes_html(self):
        html = render_index([entry("x", title="<скрипт> &")])
        self.assertNotIn("<скрипт>", html)
        self.assertIn("&lt;скрипт&gt; &amp;", html)

    def test_newest_first(self):
        html = render_index([
            entry("old", updated="2026-01-01"),
            entry("new", updated="2026-02-02"),
        ])
        self.assertLess(html.index('href="pages/new/"'), html.index('href="pages/old/"'))

    def test_empty_gallery(self):
        self.assertIn("Пока нет ни одной страницы", render_index([]))

    def test_deterministic_regardless_of_input_order(self):
        entries = [entry("a", title="A"), entry("b", kind="file")]
        self.assertEqual(render_index(entries), render_index(list(reversed(entries))))

    def test_page_href(self):
        self.assertEqual(page_href(entry("a")), "pages/a/")
        self.assertEqual(page_href(entry("a/b", kind="file")), "pages/a/b")


if __name__ == "__main__":
    unittest.main()

def page_href(entry: PageEntry) -> str:
    suffix = "/" if entry.kind == "folder" else ""
    return f"pages/{entry.key}{suffix}"


def sort_entries(entries: list[PageEntry]) -> list[PageEntry]:
    by_key = sorted(entries, key=lambda e: e.key)
    return sorted(by_key, key=lambda e: e.updated, reverse=True)


def _card(entry: PageEntry) -> str:
    return (CARD
            .replace("__HREF__", escape(page_href(entry)))
            .replace("__TITLE__", escape(entry.title or entry.key))
            .replace("__KEY__", escape(entry.key))
            .replace("__DESC__", escape(entry.description) or "—")
            .replace("__UPDATED__", escape(entry.updated)))


def render_index(entries: list[PageEntry]) -> str:
    cards = "\n".join(_card(e) for e in sort_entries(entries)) if entries else EMPTY
    return (TEMPLATE
            .replace("__COUNT__", str(len(entries)))
            .replace("__CARDS__", cards)
            .replace("__TODAY__", date.today().isoformat()))
```

import os
import shutil
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

SCRIPT = Path(__file__).resolve().parents[1] / "rebuild_index.py"


class CliTests(unittest.TestCase):
    def setUp(self):
        self.tmp = Path(tempfile.mkdtemp())
        self.addCleanup(shutil.rmtree, self.tmp, ignore_errors=True)

    def run_cli(self, *args: str) -> subprocess.CompletedProcess:
        return subprocess.run(
            [sys.executable, str(SCRIPT), *args],
            cwd=self.tmp, capture_output=True, text=True, encoding="utf-8",
            env={**os.environ, "PYTHONIOENCODING": "utf-8"},
        )

    def write_page(self, rel: str, text: str) -> None:
        p = self.tmp / "pages" / rel
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(text, encoding="utf-8")

    def test_exit1_without_pages_dir(self):
        self.assertEqual(self.run_cli().returncode, 1)

    def test_rebuild_creates_index(self):
        self.write_page("a/index.html", "<!-- page-meta\ntitle: A\n--><html></html>")
        r = self.run_cli()
        self.assertEqual(r.returncode, 0, r.stderr)
        index = (self.tmp / "index.html").read_text(encoding="utf-8")
        self.assertIn('href="pages/a/"', index)
        self.assertIn(">A</a>", index)

    def test_list_mode(self):
        self.write_page("a/index.html", "<!-- page-meta\ntitle: A\n--><html></html>")
        self.write_page("b.html", "<html></html>")
        r = self.run_cli("--list")
        self.assertEqual(r.returncode, 0, r.stderr)
        lines = r.stdout.strip().splitlines()
        self.assertEqual(len(lines), 2)
        self.assertTrue(lines[0].startswith("a | A | "), lines[0])
        self.assertTrue(lines[1].startswith("b | b | "), lines[1])

#!/usr/bin/env python3
"""Пересборка галереи index.html. Запускать из корня репозитория."""
import argparse
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))

from pageslib.gallery import render_index
from pageslib.scanner import scan_pages


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Пересборка галереи git-pages")
    parser.add_argument("--list", action="store_true",
                        help="напечатать «ключ | заголовок | обновлена» и выйти")
    args = parser.parse_args(argv)
    repo_root = Path.cwd()
    try:
        entries = scan_pages(repo_root)
    except FileNotFoundError as e:
        print(f"ошибка: {e}", file=sys.stderr)
        return 1
    except OSError:
        print("ошибка: git недоступен", file=sys.stderr)
        return 2
    if args.list:
        for e in sorted(entries, key=lambda e: e.key):
            print(f"{e.key} | {e.title or e.key} | {e.updated}")
        return 0
    html = render_index(entries)
    (repo_root / "index.html").write_text(html, encoding="utf-8", newline="\n")
    print(f"галерея пересобрана: страниц — {len(entries)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
